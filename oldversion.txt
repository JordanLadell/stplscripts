// @name         Vision - Live Breakpack Picker TV Dashboard

(() => {
  'use strict';

  const CONFIG = {
    apiPath: '/vision/clientWeb/flr/api/associateProductivity/',
    refreshMs: 30000,
    timeoutMs: 90000,
    cardsPerPage: 16,
    pageRotateMs: 12000,
    activeMinutes: 20,
    breakpackJobFunctionValues: [2, 3],
    secondShiftCutoff: '12:45',
    operationalDayStartHour: 4,
    paceSheetBaseLph: 95,
    useAdjustedUnits: true,
    settingsKey: 'visionLivePickerDashboardSettingsV1',
    zoneUrl: 'http://lcvyprwbv05.staples.com:8801/Home/ZoneManagement?order=recommended&includePTC=false',
    hubStatusUrl: 'http://lcvyprwbv05.staples.com:8801/Home/HubStatusDP',
    unpickedLinesUrl: 'http://lcvyprwbv05.staples.com:8801/Home/UnpickedLines',
    zoneRefreshMs: 10000
  };

  const APP_ID = 'vision-live-breakpack-tv';
  const state = {
    token: '', data: null, pickers: [], page: 0, sort: 'name', filter: '',
    running: false, refreshing: false, visionRequestId: 0, visionController: null, loadingDate: null, overlay: null, frame: null, ui: null,
    refreshTimer: null, rotateTimer: null, lastRefresh: null, lastDuration: 0,
    lastError: '', requestCount: 0, responseBytes: 0, selectedDate: '',
    zones: [], zoneError: '', zoneLastRefresh: null, zoneTimer: null, kpiRefreshBusy: false, kpiError: '', kpiLastRefresh: null, kpis: { modsCartons: null, inZoneCartons: null, linesRemaining: null, efficiency: null }, showZones: true, rankingMode: 'minutes', rankingAmount: 60, rankingPeople: 6, tickerItems: null,
    cardText: null, sidebarText: null, autoFitCards: true, sidebarWidthPercent: 22, conveyorBaselinePercent: 45.24, cardScale: 1, resizeObserver: null, rankingResults: [], rankingStart: null, rankingEnd: null, rankingError: '',
    operation: 'breakpack', zonePaceOverrides: { 'ZONE 106': 100, 'WA35': 100 }, detectedPaceAreas: [], breakpackGroups: new Set(['1', '2']), excludedUsernames: new Set(), minimumUnits: 0, cardSizing: null, autoRotate: true, pageRotateSeconds: 12, refreshSeconds: 30, userGroups: [], userGroupAssignments: {}
  };

  const esc = value => String(value ?? '').replace(/[&<>"']/g, char => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[char]));
  const n = value => Number.isFinite(Number(value)) ? Number(value) : 0;
  const fmt = value => Math.round(n(value)).toLocaleString();
  const hrs = value => n(value).toFixed(2);
  const pct = value => Number.isFinite(Number(value)) ? `${Math.round(Number(value))}%` : '--';
  const paceText = value => Number.isFinite(Number(value)) ? String(Math.round(Number(value))) : '--';
  const formatLocalDate = date => {
    const d = new Date(date);
    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
  };
  const operationalDate = (date = new Date()) => {
    const d = new Date(date);
    if (d.getHours() < CONFIG.operationalDayStartHour) d.setDate(d.getDate() - 1);
    return formatLocalDate(d);
  };
  const localDate = () => operationalDate();
  const operationalWindow = (dateText = operationalDate()) => {
    const start = new Date(`${dateText}T00:00:00`);
    start.setHours(CONFIG.operationalDayStartHour, 0, 0, 0);
    const end = new Date(start);
    end.setDate(end.getDate() + 1);
    return { start, end };
  };
  const parseDate = value => {
    if (!value) return null;
    const d = new Date(String(value).replace(' ', 'T'));
    return Number.isNaN(d.getTime()) ? null : d;
  };
  const timeText = value => {
    const d = value instanceof Date ? value : parseDate(value);
    return d ? d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) : '--';
  };
  const minutesAgo = date => date ? Math.max(0, Math.floor((Date.now() - date.getTime()) / 60000)) : 99999;
  const formatDuration = seconds => {
    const totalMinutes = Math.max(0, Math.round(n(seconds) / 60));
    const h = Math.floor(totalMinutes / 60), m = totalMinutes % 60;
    return h ? `${h}h ${String(m).padStart(2, '0')}m` : `${m}m`;
  };
  const isTodayReport = () => (state.selectedDate || operationalDate()) === operationalDate();
  function currentBreakInfo(allDetails) {
    if (!isTodayReport() || !Array.isArray(allDetails) || !allDetails.length) return null;
    const timestamp = detail => parseDate(detail?.end)?.getTime() || parseDate(detail?.start)?.getTime() || 0;
    const ordered = [...allDetails].sort((a, b) => timestamp(a) - timestamp(b));
    const latestActivity = ordered.at(-1);
    const breakRecords = ordered.filter(detail => {
      const label = String(detail?.label || detail?.job_function?.display_value || '').toLowerCase();
      return label.includes('lunch') || label.includes('break');
    });
    const latestBreak = breakRecords.at(-1);
    if (!latestActivity || !latestBreak || latestBreak !== latestActivity) return null;
    const label = String(latestBreak.label || latestBreak.job_function?.display_value || '').toLowerCase();
    const type = label.includes('lunch') ? 'Lunch' : 'Break';
    const start = parseDate(latestBreak.start);
    const end = parseDate(latestBreak.end);
    const seconds = Math.max(0, n(latestBreak.total_seconds) || (start && end ? (end - start) / 1000 : 0));
    return { type, elapsed: Math.max(0, Math.round(seconds / 60)) };
  }
  function autoShiftFromFirstPick(details) {
    const starts=(details||[]).map(d=>parseDate(d.start)).filter(Boolean).sort((a,b)=>a-b);
    if(!starts.length)return null;
    const first=starts[0], [hour,minute]=CONFIG.secondShiftCutoff.split(':').map(Number);
    const cutoff=new Date(first); cutoff.setHours(hour,minute,0,0);
    return {number:first<cutoff?'1':'2',label:first<cutoff?'First Shift':'Second Shift',firstPick:first};
  }
  function captureToken(value) {
    if (!value) return;
    const text = String(value);
    const match = text.match(/^Token\s+(.+)$/i);
    if (match && match[1].length > 10) state.token = match[1].trim();
  }

  const nativeFetch = window.fetch.bind(window);
  window.fetch = function(input, init = {}) {
    try {
      const headers = new Headers(init.headers || (input instanceof Request ? input.headers : undefined));
      captureToken(headers.get('authorization'));
    } catch (_) {}
    return nativeFetch(input, init);
  };

  const nativeSetRequestHeader = XMLHttpRequest.prototype.setRequestHeader;
  XMLHttpRequest.prototype.setRequestHeader = function(name, value) {
    if (String(name).toLowerCase() === 'authorization') captureToken(value);
    return nativeSetRequestHeader.apply(this, arguments);
  };

  function scanStorageForToken() {
    for (const store of [window.localStorage, window.sessionStorage]) {
      try {
        for (let i = 0; i < store.length; i++) {
          const key = store.key(i);
          const value = store.getItem(key) || '';
          captureToken(value);
          if (!state.token && /token|auth/i.test(key)) {
            try {
              const parsed = JSON.parse(value);
              const stack = [parsed];
              while (stack.length && !state.token) {
                const item = stack.pop();
                if (typeof item === 'string') captureToken(item);
                else if (item && typeof item === 'object') Object.values(item).forEach(v => stack.push(v));
              }
            } catch (_) {
              if (/^[a-f0-9]{30,}$/i.test(value)) state.token = value;
            }
          }
        }
      } catch (_) {}
    }
  }

  function buildApiUrl(date) {
    const url = new URL(CONFIG.apiPath, location.origin);
    url.searchParams.append('date_range', date);
    url.searchParams.append('date_range', date);
    url.searchParams.set('count', '1');
    return url;
  }

  async function fetchVision(date, controller = new AbortController(), timeoutMs = CONFIG.timeoutMs) {
    scanStorageForToken();
    const timeout = setTimeout(() => controller.abort(), timeoutMs);
    const headers = { accept: 'application/json, text/plain, */*' };
    if (state.token) headers.authorization = `Token ${state.token}`;
    const started = performance.now();
    try {
      const response = await nativeFetch(buildApiUrl(date), {
        method: 'GET', credentials: 'include', cache: 'no-store', headers, signal: controller.signal
      });
      if (!response.ok) {
        if (response.status === 401 || response.status === 403) {
          throw new Error('Vision authorization was not captured. Refresh the Associate Productivity page once, then reopen the dashboard.');
        }
        throw new Error(`Vision API returned HTTP ${response.status}`);
      }
      const text = await response.text();
      state.responseBytes = new Blob([text]).size;
      const json = JSON.parse(text);
      if (!Array.isArray(json.associateDetails)) throw new Error('associateDetails was not found in the Vision response.');
      state.lastDuration = Math.round(performance.now() - started);
      return json;
    } finally {
      clearTimeout(timeout);
    }
  }


  function isBreakpack(detail) {
    const jobValue = Number(detail?.job_function?.value);
    const jobName = String(detail?.job_function?.display_value || '')
      .trim()
      .toLowerCase();

    return (
      CONFIG.breakpackJobFunctionValues.includes(jobValue) ||
      jobName === 'carton picking' ||
      jobName === 'cart picking'
    );
  }

  function normalizedPaceArea(detail) {
    return String(detail?.work_location || '').trim().replace(/\s+/g, ' ').toUpperCase();
  }
  function isModsProduction(detail) {
    const jobName = String(detail?.job_function?.display_value || '').trim().toLowerCase();
    const area = normalizedPaceArea(detail);
    return jobName === 'carton picking' && (/^ZONE\s+\d{3}$/.test(area) || area === 'WA35');
  }
  function zoneOverrideRate(detailOrArea) {
    const area = typeof detailOrArea === 'string'
      ? String(detailOrArea).trim().replace(/\s+/g, ' ').toUpperCase()
      : normalizedPaceArea(detailOrArea);
    const value = Number(state.zonePaceOverrides?.[area]);
    return Number.isFinite(value) && value > 0 ? value : null;
  }
  function detailWeightedLines(detail, ratio = 1) {
    const seconds = Math.max(0, n(detail?.working_seconds)) * ratio;
    const override = zoneOverrideRate(detail);
    if (override !== null) return (seconds / 3600) * override;
    return Math.max(0, n(detail?.earned_hours)) * ratio * CONFIG.paceSheetBaseLph;
  }
  function selectedWork(detail) {
    return isModsProduction(detail);
  }

  function operationName() {
    return 'MODs';
  }

  function unitLabel() {
    return 'Lines';
  }

  function summarizeAssociate(associate) {
    const selectedDay = state.selectedDate || operationalDate();
    const { start: operationalStart, end: operationalEnd } = operationalWindow(selectedDay);
    const allSource = Array.isArray(associate.time_details) ? associate.time_details : [];
    const all = allSource.filter(detail => {
      const detailStart = parseDate(detail.start);
      const detailEnd = parseDate(detail.end) || detailStart;
      return detailStart && detailEnd && detailEnd > operationalStart && detailStart < operationalEnd;
    });
    const bulk = all.filter(isModsProduction);
    if (!bulk.length) return null;
    const orderedAll = [...all].sort((a, b) => (parseDate(a.end)?.getTime() || 0) - (parseDate(b.end)?.getTime() || 0));
    const orderedBulk = [...bulk].sort((a, b) => (parseDate(a.end)?.getTime() || 0) - (parseDate(b.end)?.getTime() || 0));
    const latestOverall = orderedAll.at(-1) || null;
    const latestBulk = orderedBulk.at(-1) || null;
    const rawUnits = bulk.reduce((sum, d) => sum + n(d.unit_quantity), 0);
    const adjusted = bulk.reduce((sum, d) => sum + n(d.adjusted_unit_quantity), 0);
    const units = adjusted;
    const seconds = bulk.reduce((sum, d) => sum + n(d.working_seconds), 0);
    const earned = bulk.reduce((sum, d) => sum + n(d.earned_hours), 0);
    const weightedLines = bulk.reduce((sum, d) => sum + detailWeightedLines(d), 0);
    const rate = seconds > 0 ? units / (seconds / 3600) : 0;
    const adjustedRate = seconds > 0 ? adjusted / (seconds / 3600) : 0;
    const performance = seconds > 0 && earned > 0 ? earned / (seconds / 3600) * 100 : null;
    const areas = {};
    for (const d of bulk) {
      const area = String(d.work_location || 'UNASSIGNED').trim().toUpperCase();
      if (!areas[area]) areas[area] = { area, units: 0, adjusted: 0, seconds: 0, earned: 0, records: 0 };
      areas[area].units += n(d.adjusted_unit_quantity);
      areas[area].adjusted += n(d.adjusted_unit_quantity);
      areas[area].seconds += n(d.working_seconds);
      areas[area].earned += n(d.earned_hours);
      areas[area].records++;
    }
    const offStandard = { approvedSeconds: 0, pendingSeconds: 0, partialSeconds: 0, deniedSeconds: 0, deniedPartialSeconds: 0, totalSeconds: 0, records: [] };
    for (const d of all) {
      if (!d.offstandard) continue;
      const statusName = String(d.offstandard.status || '').toUpperCase();
      const totalSeconds = Math.max(0, n(d.total_seconds));
      const approvedSeconds = Math.max(0, n(d.offstandard.approved_time) * 60);
      offStandard.totalSeconds += totalSeconds;
      offStandard.records.push(d);
      if (statusName === 'APPROVED') offStandard.approvedSeconds += totalSeconds;
      else if (statusName === 'PENDING' || statusName === 'CREATED') offStandard.pendingSeconds += totalSeconds;
      else if (statusName === 'DENIED') offStandard.deniedSeconds += totalSeconds;
      else if (statusName === 'PARTIAL') {
        const approvedPartialSeconds = Math.min(totalSeconds, approvedSeconds);
        const deniedPartialSeconds = Math.max(0, totalSeconds - approvedPartialSeconds);
        offStandard.partialSeconds += totalSeconds;
        offStandard.approvedSeconds += approvedPartialSeconds;
        offStandard.deniedPartialSeconds += deniedPartialSeconds;
      }
    }
    const deniedPenaltySeconds = offStandard.deniedSeconds + offStandard.deniedPartialSeconds;
    const paceSeconds = seconds;
    const paceRate = paceSeconds > 0 ? units / (paceSeconds / 3600) : 0;
    const pacePerformance = paceSeconds > 0 ? weightedLines / (paceSeconds / 3600) : null;
    const latestEnd = parseDate(latestOverall?.end);
    let status = latestOverall?.label || latestOverall?.job_function?.display_value || 'No Status';
    if (latestOverall && selectedWork(latestOverall)) status = 'Breakpack Picking';
    const offStatus = latestOverall?.offstandard?.status || '';
    const breakInfo = currentBreakInfo(all);
    const autoShift = autoShiftFromFirstPick(bulk);
    return {
      id: associate.id, label: associate.label || associate.full_name || String(associate.employee_id),
      fullName: associate.full_name || '', employeeId: associate.employee_id || '',
      units, adjusted, seconds, earned, rate, adjustedRate, performance,
      paceRate, pacePerformance, weightedLines, deniedPenaltySeconds, offStandard,
      latestOverall, latestBulk, latestEnd, currentArea: latestBulk?.work_location || 'UNASSIGNED',
      status, offStatus, breakInfo, autoShift, active: minutesAgo(latestEnd) <= CONFIG.activeMinutes,
      bulk, all, areas: Object.values(areas).sort((a, b) => b.units - a.units)
    };
  }

  function processData(json) {
    state.data = json;
    const detected = new Set(Object.keys(state.zonePaceOverrides || {}));
    for (const associate of (json.associateDetails || [])) {
      for (const detail of (associate.time_details || [])) {
        if (isModsProduction(detail)) detected.add(normalizedPaceArea(detail));
      }
    }
    state.detectedPaceAreas = [...detected].sort((a, b) => a.localeCompare(b, undefined, { numeric: true }));
    const excluded = new Set([...state.excludedUsernames].map(value => value.toLowerCase()));
    state.pickers = json.associateDetails.filter(a => !excluded.has(String(a.label || '').trim().toLowerCase())).map(summarizeAssociate).filter(p => p && p.autoShift && state.breakpackGroups.has(p.autoShift.number) && p.units >= state.minimumUnits);
    state.lastRefresh = new Date();
    state.requestCount++;
    state.lastError = '';
  }

  function sortedPickers() {
    const query = state.filter.trim().toLowerCase();
    const list = state.pickers.filter(p => !query || `${p.label} ${p.fullName} ${p.currentArea}`.toLowerCase().includes(query));
    list.sort((a, b) => {
      if (state.sort === 'recent') return (b.latestEnd?.getTime() || 0) - (a.latestEnd?.getTime() || 0);
      if (state.sort === 'rate') return b.paceRate - a.paceRate;
      if (state.sort === 'performance') return n(b.pacePerformance) - n(a.pacePerformance);
      if (state.sort === 'name') return a.label.localeCompare(b.label);
      return (CONFIG.useAdjustedUnits ? b.adjusted - a.adjusted : b.units - a.units);
    });
    return list;
  }

  async function refresh(requestedDate = null, options = {}) {
    if (!state.running && !options.force) return;
    const targetDate = requestedDate || state.ui?.date?.value || state.selectedDate || localDate();
    const historical = targetDate !== operationalDate();

    if (options.supersede && state.visionController) {
      try { state.visionController.abort(); } catch (_) {}
    } else if (state.refreshing) {
      return;
    }

    const requestId = ++state.visionRequestId;
    const controller = new AbortController();
    state.visionController = controller;
    state.refreshing = true;
    state.loadingDate = targetDate;
    state.lastError = '';
    renderStatus();
    renderCards();

    try {
      const json = await fetchVision(targetDate, controller, historical ? 90000 : CONFIG.timeoutMs);
      if (requestId !== state.visionRequestId) return;

      const selectedNow = state.ui?.date?.value || state.selectedDate || localDate();
      if (selectedNow !== targetDate) return;

      state.selectedDate = targetDate;
      processData(json);
      state.loadingDate = null;
      render();
    } catch (error) {
      if (requestId !== state.visionRequestId) return;
      state.loadingDate = null;
      if (error.name !== 'AbortError') {
        state.lastError = error.message || String(error);
      } else if (!options.supersede) {
        state.lastError = `Vision request for ${targetDate} timed out.`;
      }
      render();
    } finally {
      if (requestId === state.visionRequestId) {
        state.refreshing = false;
        state.loadingDate = null;
        state.visionController = null;
        renderStatus();
        renderCards();
      }
    }
  }
  function restartDashboardTimers() {
    clearInterval(state.refreshTimer);
    clearInterval(state.rotateTimer);
    clearInterval(state.zoneTimer);
    state.refreshTimer = null;
    state.rotateTimer = null;
    if (!state.running) return;
    state.refreshTimer = setInterval(() => {
      if ((state.ui?.date?.value || state.selectedDate) === operationalDate()) refresh();
    }, state.refreshSeconds * 1000);
    if (state.autoRotate) {
      state.rotateTimer = setInterval(() => {
        if (!state.running || !state.ui || state.ui.detailOpen) return;
        const pageSize = Math.max(1, (state.cardSizing?.columns || 4) * (state.cardSizing?.rows || 4));
        const pages = Math.max(1, Math.ceil(sortedPickers().length / pageSize));
        if (pages <= 1) return;
        state.page = (state.page + 1) % pages;
        renderCards();
      }, state.pageRotateSeconds * 1000);
    }
  }
  function start() {
    if (state.running) return;
    state.running = true;
    refresh(state.ui?.date?.value || state.selectedDate, { force: true, supersede: true });
    restartDashboardTimers();
    restartZoneTimer();
    renderStatus();
  }

  function stop() {
    state.running = false;
    state.refreshing = false;
    state.loadingDate = null;
    if (state.visionController) { try { state.visionController.abort(); } catch (_) {} }
    state.visionController = null;
    state.visionRequestId++;
    clearInterval(state.refreshTimer);
    clearInterval(state.rotateTimer);
    state.refreshTimer = null;
    state.rotateTimer = null;
    state.zoneTimer = null;
    renderStatus();
  }

  function performanceClass(value) {
    if (!Number.isFinite(Number(value))) return 'neutral';
    if (value >= 100) return 'good';
    if (value >= 85) return 'warn';
    return 'bad';
  }

  function renderStatus() {
    if (!state.ui) return;
    const parts = [state.running ? (state.refreshing ? 'REFRESHING' : 'LIVE') : 'STOPPED'];
    parts.push(`${state.loadingDate ? 'Loading' : 'Operational Date'}: ${state.loadingDate || state.selectedDate || operationalDate()} (4:00 AM reset)`);
    if (state.lastRefresh) parts.push(`Updated: ${state.lastRefresh.toLocaleTimeString()}`);
    if (state.lastDuration) parts.push(`API: ${state.lastDuration} ms`);
    if (state.responseBytes) parts.push(`Response: ${(state.responseBytes / 1024 / 1024).toFixed(1)} MB`);
    if (state.lastError) parts.push(`ERROR: ${state.lastError}`);
    state.ui.status.textContent = parts.join(' | ');
    state.ui.start.disabled = state.running;
    state.ui.stop.disabled = !state.running;
  }

  function shiftSummary(shiftNumber) {
    const pickers = state.pickers.filter(picker => picker.autoShift?.number === String(shiftNumber));
    const lines = pickers.reduce((sum, picker) => sum + picker.units, 0);
    const productiveSeconds = pickers.reduce((sum, picker) => sum + picker.seconds, 0);
    const deniedSeconds = pickers.reduce((sum, picker) => sum + picker.deniedPenaltySeconds, 0);
    const earnedHours = pickers.reduce((sum, picker) => sum + picker.earned, 0);
    const weightedLines = pickers.reduce((sum, picker) => sum + picker.weightedLines, 0);
    const adjustedSeconds = productiveSeconds;
    return {
      pickers: pickers.length,
      active: isTodayReport() ? pickers.filter(picker => picker.active).length : null,
      lines,
      pace: adjustedSeconds > 0 ? lines / (adjustedSeconds / 3600) : 0,
      performance: adjustedSeconds > 0 ? weightedLines / (adjustedSeconds / 3600) : null
    };
  }
  function renderTicker() {
    if (!state.ui) return;
    const list = state.pickers;
    const first = shiftSummary('1');
    const second = shiftSummary('2');
    const active = isTodayReport() ? list.filter(picker => picker.active).length : 'N/A';
    const topThree = [...list]
      .filter(picker => Number.isFinite(Number(picker.pacePerformance)))
      .sort((a, b) => b.pacePerformance - a.pacePerformance || b.units - a.units)
      .slice(0, 3);

    const enabled=state.tickerItems||DEFAULT_TICKER_ITEMS;
    const items = [
      enabled.firstPerf && ['1ST SHIFT AVG LPH', paceText(first.performance)],
      enabled.secondPerf && ['2ND SHIFT AVG LPH', paceText(second.performance)],
      enabled.active && ['TOTAL ACTIVE PICKERS', active],
      enabled.top1 && ['TOP ADJUSTED LPH', topThree[0] ? `${topThree[0].label} ${paceText(topThree[0].pacePerformance)}` : '--'],
      enabled.top2 && ['2ND ADJUSTED LPH', topThree[1] ? `${topThree[1].label} ${paceText(topThree[1].pacePerformance)}` : '--'],
      enabled.top3 && ['3RD ADJUSTED LPH', topThree[2] ? `${topThree[2].label} ${paceText(topThree[2].pacePerformance)}` : '--'],
      enabled.firstPicks && ['1ST SHIFT PICKS', fmt(first.lines)],
      enabled.secondPicks && ['2ND SHIFT PICKS', fmt(second.lines)]
    ].filter(Boolean);

    const itemHtml = items
      .map(([label, value]) => `<div class="tickerItem"><b>${esc(label)}</b><strong>${esc(value)}</strong></div>`)
      .join('');

    state.ui.ticker.innerHTML = `<div class="tickerTrack"><div class="tickerSet">${itemHtml}</div><div class="tickerSet" aria-hidden="true">${itemHtml}</div></div>`;
  }
  function isNormalNumberedZone(detail) {
    return isModsProduction(detail);
  }
  function rankingWindow() {
    const end=new Date(), {start:dayStart}=operationalWindow(operationalDate(end));
    if(state.rankingMode==='today')return {start:dayStart,end,label:'TODAY OVERALL'};
    const amount=state.rankingMode==='hours'?state.rankingAmount*3600000:state.rankingAmount*60000;
    const unit=state.rankingMode==='hours'?(state.rankingAmount===1?'HOUR':'HOURS'):(state.rankingAmount===1?'MINUTE':'MINUTES');
    return {start:new Date(Math.max(dayStart.getTime(),end.getTime()-amount)),end,label:`LAST ${state.rankingAmount} ${unit}`};
  }
  function calculateLastHourRankings() {
    if(!isTodayReport())return [];
    const {start:windowStart,end:windowEnd}=rankingWindow();
    return state.pickers.map(picker=>{
      let units=0,productiveSeconds=0,earned=0,weightedLines=0,deniedSeconds=0;
      for(const d of picker.all.filter(isNormalNumberedZone)){
        const start=parseDate(d.start),end=parseDate(d.end);if(!start||!end||end<=windowStart||start>=windowEnd)continue;
        const duration=Math.max(1,(end-start)/1000), overlap=Math.max(0,(Math.min(end,windowEnd)-Math.max(start,windowStart))/1000), ratio=Math.min(1,overlap/duration);
        units+=n(d.adjusted_unit_quantity)*ratio;
        productiveSeconds+=n(d.working_seconds)*ratio;
        earned+=n(d.earned_hours)*ratio;
        weightedLines+=detailWeightedLines(d,ratio);
      }
      for(const d of picker.all.filter(detail=>detail?.offstandard)){
        const start=parseDate(d.start),end=parseDate(d.end);if(!start||!end||end<=windowStart||start>=windowEnd)continue;
        const duration=Math.max(1,(end-start)/1000), overlap=Math.max(0,(Math.min(end,windowEnd)-Math.max(start,windowStart))/1000), ratio=Math.min(1,overlap/duration);
        const totalSeconds=Math.max(0,n(d.total_seconds))*ratio;
        const status=String(d.offstandard?.status||'').toUpperCase();
        if(status==='DENIED') deniedSeconds+=totalSeconds;
        else if(status==='PARTIAL'){
          const approvedSeconds=Math.max(0,n(d.offstandard?.approved_time)*60)*ratio;
          deniedSeconds+=Math.max(0,totalSeconds-approvedSeconds);
        }
      }
      const adjustedSeconds=productiveSeconds;
      return {label:picker.label,lines:units,productiveSeconds,deniedSeconds,performance:adjustedSeconds>0?weightedLines/(adjustedSeconds/3600):null};
    }).filter(x=>Number.isFinite(x.performance)).sort((a,b)=>b.performance-a.performance||b.lines-a.lines);
  }

  function parseZoneManagementHtml(html) {
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const heading = [...doc.querySelectorAll('h1,h2,h3')].some(node => /zone management/i.test(node.textContent || ''));
    const table = [...doc.querySelectorAll('table')].find(candidate => /rec\.\s*pickers/i.test(candidate.tHead?.textContent || ''));
    if (!heading || !table) throw new Error('LOGIN_REQUIRED');
    const headers = [...table.querySelectorAll('thead th')].map(th => th.textContent.replace(/\s+/g, ' ').trim().toLowerCase());
    const index = name => headers.findIndex(header => header === name || header.includes(name));
    const indexes = { location:index('location'), inZone:index('in zone'), inactive:index('inactive'), traveling:index('traveling'), packsize:index('packsize'), induction:index('induction'), pickers:index('pickers'), qty:index('pickers quantity'), recommended:index('rec. pickers') };
    return [...table.querySelectorAll('tbody tr')].map(row => {
      const cells=[...row.cells];
      const text = key => indexes[key] >= 0 ? (cells[indexes[key]]?.textContent || '').replace(/\s+/g,' ').trim() : '';
      const num = key => Number(text(key).replace(/,/g,'')) || 0;
      const location=text('location');
      if (!location || /^total$/i.test(location)) return null;
      const pickerQty=num('qty'), recommended=num('recommended');
      const cleanLocation=location.replace(/^M(?=\d)/i,'');
      const cartonsInZone=num('inZone')+num('inactive');
      const cartonsOnTheWay=num('traveling')+num('packsize')+num('induction');
      return {location:cleanLocation,cartonsInZone,cartonsOnTheWay,totalCartons:cartonsInZone+cartonsOnTheWay,pickers:text('pickers'),pickerQty,recommended,need:recommended-pickerQty};
    }).filter(Boolean).sort((a,b)=>b.need-a.need||b.totalCartons-a.totalCartons||a.location.localeCompare(b.location,undefined,{numeric:true}));
  }

  function parseHubStatusHtml(html) {
    const doc=new DOMParser().parseFromString(html,'text/html');
    const table=[...doc.querySelectorAll('table')].find(t=>/\bmod\b/i.test(t.tHead?.textContent||'')&&/\btotal\b/i.test(t.tHead?.textContent||''));
    if(!table)throw new Error('LOGIN_REQUIRED');
    const headers=[...table.querySelectorAll('thead th')].map(th=>th.textContent.replace(/\s+/g,' ').trim().toLowerCase());
    const modIndex=headers.findIndex(h=>h==='mod');
    if(modIndex<0)throw new Error('MOD_COLUMN_NOT_FOUND');
    const totalRow=[...table.querySelectorAll('tbody tr')].find(row=>/^total$/i.test((row.cells[0]?.textContent||'').trim()));
    if(!totalRow)throw new Error('HUB_TOTAL_NOT_FOUND');
    return Number((totalRow.cells[modIndex]?.textContent||'0').replace(/[^0-9.-]/g,''))||0;
  }
  function parseUnpickedLinesHtml(html) {
    const doc=new DOMParser().parseFromString(html,'text/html');
    const text=doc.body?.textContent||'';
    const match=text.match(/Total\s+Available\s+Lines\s*:\s*([\d,]+)/i);
    if(match)return Number(match[1].replace(/,/g,''))||0;

    const totalRow=table?[...table.querySelectorAll('tbody tr')].find(r=>/^total$/i.test((r.cells[0]?.textContent||'').trim())):null;

    throw new Error('UNPICKED_TOTAL_NOT_FOUND');
  }

  function readableTextColor(hex) {
    const value = normalizeGroupColor(hex).slice(1);
    const r = parseInt(value.slice(0, 2), 16), g = parseInt(value.slice(2, 4), 16), b = parseInt(value.slice(4, 6), 16);

  function normalizeUserGroups(rawGroups = []) {
    const groups = [];
    const usedIds = new Set();
    const usedNames = new Set();
    for (const item of Array.isArray(rawGroups) ? rawGroups : []) {
      const name = normalizeGroupName(typeof item === 'string' ? item : item?.name);
      if (!name || usedNames.has(name.toLowerCase())) continue;
      let id = String(typeof item === 'string' ? '' : item?.id || '').trim();
      if (!id || usedIds.has(id)) id = makeGroupId();
      const abbreviation = normalizeGroupAbbreviation(typeof item === 'string' ? '' : item?.abbreviation, name);
      const color = normalizeGroupColor(typeof item === 'string' ? '' : item?.color);
      groups.push({ id, name, abbreviation, color });
      usedIds.add(id);
      usedNames.add(name.toLowerCase());
    }
    return groups;
  }
  function normalizeGroupAssignments(rawAssignments = {}, rawGroups = state.userGroups || []) {
    const groups = normalizeUserGroups(rawGroups);
    const validIds = new Set(groups.map(group => group.id));
    const assignments = {};
    if (rawAssignments && typeof rawAssignments === 'object' && !Array.isArray(rawAssignments)) {
      for (const [username, groupId] of Object.entries(rawAssignments)) {
        const normalizedUsername = String(username || '').trim().toLowerCase();
        const normalizedGroupId = String(groupId || '').trim();
        if (normalizedUsername && validIds.has(normalizedGroupId)) assignments[normalizedUsername] = normalizedGroupId;
      }
    }
    return assignments;
  }

  function normalizeZonePaceOverrides(raw = {}) {
    const source = raw && typeof raw === 'object' && !Array.isArray(raw) ? raw : {};
    const out = { 'ZONE 106': 100, 'WA35': 100 };
    for (const [area, rate] of Object.entries(source)) {
      const key = String(area || '').trim().replace(/\s+/g, ' ').toUpperCase();
      const value = Number(rate);
      if ((/^ZONE\s+\d{3}$/.test(key) || key === 'WA35') && Number.isFinite(value) && value > 0 && value <= 1000) out[key] = value;
      else if ((/^ZONE\s+\d{3}$/.test(key) || key === 'WA35') && (rate === '' || rate === null || rate === false)) delete out[key];
    }
    return out;
  }

  function loadSettings() {
    try {
      const stored = localStorage.getItem(CONFIG.settingsKey);
      const raw = stored ? JSON.parse(stored) : {};
      if (!raw.excludedUsernames && raw.excludedEmployeeIds) raw.excludedUsernames = raw.excludedEmployeeIds;
      applySettings(raw);
    } catch (error) {
      console.error('[Vision Dashboard] Could not load settings:', error);
      applySettings({});
    }
  }
  function saveSettings() {
    const settings = currentSettings();
    localStorage.setItem(CONFIG.settingsKey, JSON.stringify(settings));
    const verification = JSON.parse(localStorage.getItem(CONFIG.settingsKey) || '{}');
    if (JSON.stringify(normalizeSettings(verification)) !== JSON.stringify(settings)) {
      throw new Error('Browser storage verification failed.');
    }
    return settings;
  }

    };
    const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' });

    const link = document.createElement('a');
    link.href = url;
    link.download = 'Vision_Live_Breakpack_Dashboard_Settings.json';
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  function openDashboard() {
    if (document.getElementById(`${APP_ID}-overlay`)) return;
    state.selectedDate = operationalDate();
    const overlay = document.createElement('div');
    overlay.id = `${APP_ID}-overlay`;
    Object.assign(overlay.style, { position: 'fixed', inset: '0', zIndex: '2147483646', background: '#000' });
    const frame = document.createElement('iframe');
    frame.srcdoc = dashboardHtml();
    frame.setAttribute('sandbox', 'allow-scripts allow-same-origin');
    Object.assign(frame.style, { width: '100%', height: '100%', border: '0' });
    overlay.appendChild(frame);
    document.body.appendChild(overlay);
    state.overlay = overlay;
    state.frame = frame;
    frame.onload = () => {
      const d = frame.contentDocument;
      const $ = id => d.getElementById(id);
      state.ui = { ticker: $('ticker'), cards: $('cards'), kpis: $('kpis'), rank: $('rank'), feed: $('feed'), zones: $('zones'), zoneStamp: $('zoneStamp'), status: $('status'), page: $('page'), filter: $('filter'), sort: $('sort'), date: $('date'), groupFilter: $('groupFilter'), firstShift: $('firstShift'), secondShift: $('secondShift'), rankingTitle: $('rankingTitle'), activityTitle: $('activityTitle'), start: $('start'), stop: $('stop'), detailOpen: false };
      state.ui.date.value = state.selectedDate;
      state.ui.sort.value = state.sort;
      restartResponsiveSizingObserver();
      $('prev').onclick = () => { state.page--; renderCards(); };
      $('next').onclick = () => { state.page++; renderCards(); };
      const updateGroups = () => {
        state.breakpackGroups = new Set();
        if (state.ui.firstShift.checked) state.breakpackGroups.add('1');
        if (state.ui.secondShift.checked) state.breakpackGroups.add('2');
        state.page = 0;
        processData(state.data || { associateDetails: [] });
        render();
      };
      $('firstShift').onchange = updateGroups;
      $('secondShift').onchange = updateGroups;
      $('filter').oninput = event => { state.filter = event.target.value; state.page = 0; renderCards(); };
      $('sort').onchange = event => {
        state.sort = event.target.value;
        state.page = 0;
        saveSettings();
        renderCards();
      };
      $('date').onchange = event => {
        const selected = event.target.value;
        state.selectedDate = selected;
        state.page = 0;
        state.lastError = '';
        refresh(selected, { force: true, supersede: true });
      };
      $('refresh').onclick = () => refresh(state.ui.date.value, { force: true, supersede: true });
      $('start').onclick = start;
      $('stop').onclick = stop;
      $('settings').onclick = openSettings;
      $('fullscreen').onclick = () => overlay.requestFullscreen?.();
      $('close').onclick = closeDashboard;
      render();
      start();
    };
  }

  function closeDashboard() {
    stop();
    document.getElementById(`${APP_ID}-overlay`)?.remove();
    state.resizeObserver?.disconnect?.(); state.resizeObserver=null;
    state.overlay = null;
    state.frame = null;
    state.ui = null;
  }

  function addLaunchButton() {
    if (!document.body || document.getElementById(`${APP_ID}-button`)) return;
    const button = document.createElement('button');
    button.id = `${APP_ID}-button`;
    button.textContent = 'Open Live Breakpack TV Dashboard';
    Object.assign(button.style, {
      position: 'fixed', right: '20px', bottom: '56px', zIndex: '2147483645',
      background: '#087fc2', color: '#fff', border: '1px solid #58bcea', borderRadius: '999px',
      padding: '12px 18px', fontWeight: '900', cursor: 'pointer', boxShadow: '0 7px 22px rgba(0,0,0,.38)'
    });
    button.onclick = openDashboard;
    document.body.appendChild(button);
  }

  function initialize() {
    loadSettings();
    scanStorageForToken();
    addLaunchButton();
    const observer = new MutationObserver(addLaunchButton);
    observer.observe(document.documentElement, { childList: true, subtree: true });
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', initialize, { once: true });
  else initialize();
